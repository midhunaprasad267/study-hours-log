#include <stdio.h>
#include <string.h>

int main()
{
    char subject[50][50];
    float hours[50];
    int count = 0;
    int choice, i, found;
    char search[50];

    do
    {
        printf("\n--- STUDY HOURS LOG ---\n");
        printf("1. Add Entry\n");
        printf("2. View All Entries\n");
        printf("3. Search by Subject\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                if(count == 50)
                {
                    printf("Log is full!\n");
                }
                else
                {
                    printf("Enter subject name: ");
                    scanf(" %[^\n]", subject[count]);

                    printf("Enter study hours: ");
                    scanf("%f", &hours[count]);

                    count++;

                    printf("Entry added successfully!\n");
                }
                break;

            case 2:
                if(count == 0)
                {
                    printf("No entries available.\n");
                }
                else
                {
                    printf("\nS.No.  Subject  Hours\n");

                    for(i = 0; i < count; i++)
                    {
                        printf("%d      %s      %.2f\n",
                               i + 1, subject[i], hours[i]);
                    }
                }
                break;

            case 3:
                printf("Enter subject to search: ");
                scanf(" %[^\n]", search);

                found = 0;

                for(i = 0; i < count; i++)
                {
                    if(strcmp(subject[i], search) == 0)
                    {
                        printf("S.No.: %d\n", i + 1);
                        printf("Subject: %s\n", subject[i]);
                        printf("Hours: %.2f\n", hours[i]);

                        found = 1;
                    }
                }

                if(found == 0)
                {
                    printf("No matching entry found.\n");
                }
                break;

            case 4:
                printf("Exiting program.\n");
                break;

            default:
                printf("Invalid choice!\n");
        }

    } while(choice != 4);

    return 0;
}
